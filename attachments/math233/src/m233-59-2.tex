% Figure for Calculus §59.
\begin{tikzpicture}[x=1.05cm, y=1.05cm, font=\scriptsize]
  \begin{scope}
    \clip (-2.9,-2.9) rectangle (2.9,2.9);
    % the parabolas x = k y^2
    \foreach \k in {0.25,0.5,1,2,-0.25,-0.5,-1,-2} {
      \draw[figB, thick] plot[domain=-3:3, samples=100] ({\k*\x*\x},\x);
    }
    % the ellipses x^2 + y^2/2 = C
    \foreach \C in {0.75,1.75,3.2} {
      \draw[figR, thick] (0,0) ellipse ({sqrt(\C)} and {sqrt(2*\C)});
    }
  \end{scope}
  \draw[-stealth, thin, black!70] (-3.05,0) -- (3.15,0) node[right] {$x$};
  \draw[-stealth, thin, black!70] (0,-3.05) -- (0,3.2) node[above] {$y$};
  % right angle at (1/2, 1): parabola k = 1/2 and ellipse C = 3/4
  \fill (0.5,1) circle (1.4pt);
  \draw[thin] (0.5,1) ++(0.13,0.13) -- ++(-0.13,0.13) -- ++(-0.13,-0.13);
  \node[figB, fill=white, inner sep=1.5pt] at (2.2,2.75) {$x=ky^2$};
  \node[figR, fill=white, inner sep=1pt] at (2.35,-2.55) {$x^2+\tfrac12 y^2=C$};
\end{tikzpicture}

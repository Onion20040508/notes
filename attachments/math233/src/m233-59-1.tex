% Figure for Calculus §59.
\begin{tikzpicture}[x=1.4cm, y=0.42cm, font=\scriptsize]
  \def\L{0.12}
  \begin{scope}
    \clip (-2.1,-6.3) rectangle (2.1,6.3);
    % direction field of y' = x^2 y (segments of fixed screen length)
    \foreach \xx in {-2,-1.75,...,2.01} {
      \foreach \yy in {-6,-5,...,6} {
        \pgfmathsetmacro{\m}{\xx*\xx*\yy}
        \pgfmathsetmacro{\nn}{sqrt(1.96+(\m*0.42)^2)}
        \pgfmathsetmacro{\ux}{\L*1.4/\nn}
        \pgfmathsetmacro{\uy}{\L*\m*0.42/\nn}
        \pgfmathsetmacro{\vx}{-\ux}
        \pgfmathsetmacro{\vy}{-\uy}
        \draw[black!45, thin] (\xx,\yy) ++(\vx cm,\vy cm) -- ++(2*\ux cm,2*\uy cm);
      }
    }
    % solution curves y = A e^{x^3/3}
    \foreach \A in {3,1,0.3,-0.3,-1,-3} {
      \draw[figB, thick] plot[domain=-2:2, samples=120] (\x,{\A*exp(\x*\x*\x/3)});
    }
    \draw[figR, very thick] (-2.1,0) -- (2.1,0);
  \end{scope}
  \draw[-stealth, thin, black!70] (-2.25,0) -- (2.35,0) node[right] {$x$};
  \draw[-stealth, thin, black!70] (0,-6.5) -- (0,6.8) node[above] {$y$};
  \foreach \t in {-2,-1,1,2} {
    \draw[thin, black!70] (\t,0.15) -- (\t,-0.15);
    \node[below, fill=white, inner sep=1pt] at (\t,-0.45) {$\t$};
  }
  \foreach \t in {-6,-4,-2,2,4,6} {
    \draw[thin, black!70] (0.04,\t) -- (-0.04,\t);
    \node[left, fill=white, inner sep=1pt] at (-0.05,\t) {$\t$};
  }
  \node[figR, above right] at (1.35,0) {$y=0$};
  \node[figB, left, fill=white, inner sep=1pt] at (1.08,5.3) {$A=3$};
  \node[figB, left, fill=white, inner sep=1pt] at (1.08,-5.3) {$A=-3$};
\end{tikzpicture}

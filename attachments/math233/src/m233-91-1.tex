    % lines through the origin
    \foreach \m in {0.6,-0.6,2,-2} \draw[black!45, dashed, thin] (-2.3,{-2.3*\m}) -- (2.3,{2.3*\m});

    \foreach \l in {3,0.3333} {
      \draw[figG!70!black, thin] plot[domain=-1.65:1.65, samples=80] ({\l*\x*\x},\x);

    }
    \draw[figG, very thick] plot[domain=-1.65:1.65, samples=80] ({\x*\x},\x);
    \draw[figR, very thick] plot[domain=-1.65:1.65, samples=80] ({-\x*\x},\x);

  \node[figR, anchor=east] at (-2.32,1.52) {$f=-\tfrac12$ on $x=-y^2$};

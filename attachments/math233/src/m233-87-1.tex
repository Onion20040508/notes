% Figure for Calculus §87.
\begin{tikzpicture}[font=\scriptsize, >=Stealth, declare function={cx(\t)=\t; cy(\t)=1.2+1.6*sin(deg(0.9*\t))-0.08*\t*\t;}]
  \foreach \sh/\lab in {0/a, 6.4/b}{
  \begin{scope}[xshift=\sh cm]
    \draw[black!70, thick] plot[domain=-0.5:3.9, samples=60, smooth] ({cx(\x)},{cy(\x)});
    \coordinate (O) at (2.7,-0.5);
    \coordinate (P) at ({cx(0.3)},{cy(0.3)});
    \coordinate (Q) at ({cx(1.4)},{cy(1.4)});
    \draw[->, figG, thick] (O) -- (P) node[pos=0.55, below left] {$\mathbf{r}(t)$};
    \draw[->, figG, thick] (O) -- (Q) node[pos=0.5, right] {$\mathbf{r}(t+h)$};
    \fill (O) circle (1.2pt) node[below] {$O$};
    \fill (P) circle (1.4pt) node[above left] {$P$};
    \fill (Q) circle (1.4pt) node[below right] {$Q$};
    \node[black!70] at (3.9,{cy(3.9)+0.35}) {$C$};
  \end{scope}}
  % (a) secant
  \draw[->, figB, thick] ({cx(0.3)},{cy(0.3)}) -- ({cx(1.4)},{cy(1.4)});
  \node[figB, above left] at (0.75,{cy(0.75)+0.05}) {$\mathbf{r}(t+h)-\mathbf{r}(t)$};
  \node[font=\small] at (2.2,-1.3) {(a) the secant vector $\overrightarrow{PQ}$};
  % (b) scaled secant and tangent
  \begin{scope}[xshift=6.4cm]
    \coordinate (P) at ({cx(0.3)},{cy(0.3)}); \coordinate (Q) at ({cx(1.4)},{cy(1.4)});
    \draw[->, figB, thick] (P) -- ($(P)!1.45!(Q)$) node[anchor=south west] {$\dfrac{\mathbf{r}(t+h)-\mathbf{r}(t)}{h}$};
    % tangent direction at t=1.2: derivative
    \pgfmathsetmacro{\dy}{1.6*0.9*cos(deg(0.9*0.3))-0.16*0.3}
    \draw[->, figR, very thick] (P) -- ++({1.5/sqrt(1+\dy*\dy)},{1.5*\dy/sqrt(1+\dy*\dy)}) node[above] {$\mathbf{r}'(t)$};
    \node[font=\small] at (2.2,-1.3) {(b) the tangent vector $\mathbf{r}'(t)$};
  \end{scope}
\end{tikzpicture}

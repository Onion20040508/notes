  \def\a{3} \def\b{2} \pgfmathsetmacro{\c}{sqrt(5)}

  \draw[-stealth, thin, black!70] (-3.7,0) -- (3.9,0) node[right] {$x$};

  \draw[figR, thick] (0,0) -- (\c,0) -- (0,\b) -- cycle;

  \node[figR, left] at (0,0.55) {$b$};

  \fill (\c,0) circle (1.6pt) node[below=2pt] {$F_2(c,0)$};

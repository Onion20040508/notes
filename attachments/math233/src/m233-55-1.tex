% Figure for Calculus §55.
\begin{tikzpicture}[x=1.25cm, y=0.95cm, font=\scriptsize,
    declare function={p(\x)=1.3+3.2*exp(-\x/2.1);}]
  \def\X{4.2} \def\n{6}
  \pgfmathsetmacro{\P}{p(\X)}
  % exact consumer surplus (light) and right-endpoint rectangles
  \fill[figB!10] plot[domain=0:\X, samples=60] (\x,{p(\x)}) -- (\X,\P) -- (0,\P) -- cycle;
  \foreach \i in {1,...,6} {
    \pgfmathsetmacro{\xl}{(\i-1)*\X/\n}
    \pgfmathsetmacro{\xr}{\i*\X/\n}
    \pgfmathsetmacro{\hr}{p(\xr)}
    \fill[figB!30] (\xl,\P) rectangle (\xr,\hr);
    \draw[figB!80, thin] (\xl,\P) rectangle (\xr,\hr);
  }
  % axes
  \draw[-stealth, thin, black!70] (-0.15,0) -- (6.0,0) node[right] {$x$};
  \draw[-stealth, thin, black!70] (0,-0.15) -- (0,4.9) node[above] {$p$};
  \node[below left] at (0,0) {$0$};
  % price line and demand curve
  \draw[figR, thick] (0,\P) -- (5.6,\P) node[right] {$p = P$};
  \node[figR, left] at (0,\P) {$P$};
  \draw[figB, thick] plot[domain=0:5.6, samples=80] (\x,{p(\x)});
  \node[figB, right] at (1.0,{p(1.0)+0.25}) {$p = p(x)$};
  \fill[figB] (\X,\P) circle (1.6pt) node[above right] {$(X, P)$};
  \draw[black!35, dotted] (\X,0) -- (\X,\P);
  \draw[thin, black!70] (\X,0.06) -- (\X,-0.06) node[below] {$X$};
  % label one rectangle (i = 3)
  \pgfmathsetmacro{\xl}{2*\X/\n} \pgfmathsetmacro{\xr}{3*\X/\n} \pgfmathsetmacro{\hr}{p(\xr)}
  \draw[black!35, dotted] (\xl,0) -- (\xl,\P) (\xr,0) -- (\xr,\P);
  \node[below] at (\xl,0) {$x_{i-1}$}; \node[below] at (\xr,0) {$x_i$};
  \draw[decorate, decoration={brace, amplitude=2pt, mirror}, thin] (\xl,{\P-0.1}) -- (\xr,{\P-0.1}) node[midway, below=2pt] {$\Delta x$};
  \draw[thin] ({\xr+0.03},{\hr}) -- (2.95,2.9) node[above right, inner sep=1pt] {height $p(x_i) - P$};
\end{tikzpicture}

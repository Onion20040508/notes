% Figure for Calculus §6.
\begin{tikzpicture}[x=1.7cm, y=0.95cm, font=\scriptsize]
  \draw[-stealth, thin, black!70] (-1.3,0) -- (2.75,0) node[right] {$x$};
  \draw[-stealth, thin, black!70] (0,-1.3) -- (0,5.4) node[above] {$y$};
  \foreach \t in {1,2} \draw[thin, black!70] (\t,0.06) -- (\t,-0.06) node[below] {$\t$};
  \draw[thin, black!70] (0.04,1) -- (-0.04,1) node[left] {$1$};
  % secants through P(1,1) and Q(q,q^2): slope q+1
  \foreach \q/\s/\m in {2/45/3, 1.5/70/2.5, 1.2/100/2.2} {
    \draw[figB!\s, thick] plot[domain=0.45:2.35, samples=2] (\x,{1+(\q+1)*(\x-1)});
    \node[figB!\s, right] at (2.35,{1+(\q+1)*1.35}) {$Q$ at $x=\q$: $m_{PQ}=\m$};
  }
  \draw[figG, thick] plot[domain=-1.0:2.35, samples=2] (\x,{\x});
  \fill[figG] (0,0) circle (1.7pt);
  \node[figG, right] at (2.35,2.35) {$Q$ at $x=0$: $m_{PQ}=1$};
  % tangent y = 2x - 1
  \draw[figR, thick] plot[domain=0.25:2.35, samples=2] (\x,{2*\x-1});
  \node[figR, right] at (2.35,3.55) {tangent: $m=2$};
  % parabola
  \draw[black, thick] plot[domain=-1.2:2.25, samples=80] (\x,{\x*\x});
  \foreach \q/\s in {2/45, 1.5/70, 1.2/100} \fill[figB!\s] (\q,{\q*\q}) circle (1.8pt);
  \fill[figG] (0,0) circle (1.8pt);
  \fill[black] (1,1) circle (1.8pt);
  \node[right] at (1.04,0.72) {$P(1,1)$};
  \node[black, left] at (-1.2,1.44) {$y=x^2$};
\end{tikzpicture}

  \begin{scope}[xshift=5.2cm]

    \draw[->, figB, thick] (U) -- (S) node[midway, right] {$\mathbf{v}$};
    \draw[->, figG, thick] (O) -- (V) node[midway, left] {$\mathbf{v}$};

  \end{scope}
  % (c) Difference

    \draw[->, figR, thick] (V) -- (U) node[midway, above right] {$\mathbf{u}-\mathbf{v}$};
    \node[font=\small] at (1.4,-0.8) {(c) $\mathbf{v}+(\mathbf{u}-\mathbf{v})=\mathbf{u}$};
  \end{scope}

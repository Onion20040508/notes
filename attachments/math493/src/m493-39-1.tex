% Figure from MATH 493 notes, section 39. Compile with the notes preamble.
\begin{tikzpicture}[x=0.14cm, lab/.style={font=\scriptsize}]
  % two copies of the interval [0, 60]
  \foreach \y in {0.7,-0.7} {
    \draw[black!40, thin] (0,\y) -- (61,\y);
    \foreach \t in {0,10,...,60} \draw[black!40, thin] (\t,\y-0.06) -- (\t,\y+0.06);
  }
  % possible orders 1 + s, s a sum of some of 20, 15, 12, 12
  \foreach \v in {13,16,21,25,28,33,36,40,45,48} {
    \fill[figR] (\v,0.7) circle (1.8pt);
    \node[lab, figR, above=2pt] at (\v,0.7) {$\v$};
  }
  % divisors of 60
  \foreach \v in {2,3,4,5,6,10,15,20,30} \fill[figB] (\v,-0.7) circle (1.8pt);
  \fill[figB] (12,-0.7) circle (1.8pt);
  \foreach \v in {15,20,30} \node[lab, figB, below=2pt] at (\v,-0.7) {$\v$};
  \node[lab, figB, below=2pt] at (9.4,-0.7) {$10$};
  \node[lab, figB, below=2pt] at (12.6,-0.7) {$12$};
  \node[lab, figB, below=2pt] at (4.3,-0.7) {$2\text{--}6$};
  % the only common values
  \foreach \v in {1,60} {
    \draw[black!70, thick, rounded corners=3pt] (\v-1.3,-0.95) rectangle (\v+1.3,0.95);
    \fill[figR] (\v,0.7) circle (1.8pt);
    \fill[figB] (\v,-0.7) circle (1.8pt);
    \node[lab, above=9pt] at (\v,0.7) {$\v$};
  }
  \node[lab, figR, anchor=west] at (63,0.7) {possible $|N| = 1 + s$};
  \node[lab, figB, anchor=west] at (63,-0.7) {divisors of $60$};
\end{tikzpicture}

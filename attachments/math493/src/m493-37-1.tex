% Figure from MATH 493 notes, section 37. Compile with the notes preamble.
\begin{tikzpicture}[x=0.62cm]
  % Z with its three cosets of 3Z
  \draw[-stealth, thin, black!55] (-4.8,0) -- (7.8,0) node[right, font=\small, black] {$\mathbb{Z}$};
  \foreach \n in {-4,...,7} {
    \pgfmathtruncatemacro{\r}{mod(\n+6,3)}
    \ifcase\r \def\c{figB}\or \def\c{figR}\or \def\c{figG}\fi
    \fill[\c] (\n,0) circle (2.6pt);
    \node[font=\scriptsize, \c, below=2pt] at (\n,0) {$\n$};
  }
  \node[font=\scriptsize, black!55] at (-5.3,0) {$\cdots$};
  % the projection
  \draw[-{Stealth[length=2.2mm]}, thick, black!60] (1.5,-0.7) -- (1.5,-1.55) node[midway, right, font=\small, black] {$\pi$};
  % Z/3Z: three points, one per coset
  \begin{scope}[shift={(1.5,-3.1)}, x=1cm]
    \foreach \k/\c/\a in {0/figB/90, 1/figR/-30, 2/figG/210} {
      \node[circle, draw=\c, fill=\c!15, thick, font=\scriptsize, inner sep=1pt, minimum size=0.62cm] (q\k) at (\a:1.0) {$[\k]$};
    }
    \draw[-{Stealth[length=1.8mm]}, black!55] (q0) to[bend left=25] (q1);
    \draw[-{Stealth[length=1.8mm]}, black!55] (q1) to[bend left=25] (q2);
    \draw[-{Stealth[length=1.8mm]}, black!55] (q2) to[bend left=25] (q0);
    \node[font=\scriptsize, black!70] at (0,0) {$+[1]$};
  \end{scope}
  \node[font=\small, anchor=west] at (3.8,-3.1) {$\mathbb{Z}/3\mathbb{Z} = \{\,0 + 3\mathbb{Z},\ 1 + 3\mathbb{Z},\ 2 + 3\mathbb{Z}\,\}$};
\end{tikzpicture}

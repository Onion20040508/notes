% Figure from MATH 493 notes, section 5. Compile with the notes preamble.
\begin{tikzpicture}[x=1.45cm, y=1cm]
  % ---- case a > 0: H = aZ ----
  \begin{scope}[yshift=1.75cm]
    \def\a{1.3}
    \fill[figR!15] (0,-0.12) rectangle (\a,0.12);
    \draw[-stealth, thin] (-0.4,0) -- (4.85,0);
    \foreach \k in {0,...,3} \fill[figB] ({\k*\a},0) circle (2.3pt);
    \node[font=\scriptsize, anchor=north] at (0,-0.12) {$0$};
    \node[font=\scriptsize, anchor=north] at (\a,-0.12) {$a$};
    \node[font=\scriptsize, anchor=north] at ({2*\a},-0.12) {$2a$};
    \node[font=\scriptsize, anchor=north] at ({3*\a},-0.12) {$3a$};
    \node[font=\scriptsize, figR, anchor=south] at ({\a/2},0.12) {no point of $H$};
    \node[font=\small, figB, anchor=west] at (4.95,0) {$H = a\mathbb{Z}$};
    \node[font=\scriptsize, black!60, anchor=west] at (4.95,-0.42) {$a = \inf P > 0$};
  \end{scope}
  % ---- case a = 0: H = Z + sqrt2 Z ----
  \draw[-stealth, thin] (-0.4,0) -- (4.85,0);
  \foreach \n in {-6,...,6} {
    \foreach \m in {-10,...,14} {
      \pgfmathsetmacro{\v}{\m + \n*1.41421356}
      \pgfmathparse{(\v > -0.3) && (\v < 4.75) ? 1 : 0}
      \ifnum\pgfmathresult=1 \fill[figB] (\v,0) circle (1.25pt); \fi
    }
  }
  \foreach \k in {0,...,4} \node[font=\scriptsize, anchor=north] at (\k,-0.12) {$\k$};
  \node[font=\small, figB, anchor=west] at (4.95,0) {$H = \mathbb{Z} + \sqrt2\,\mathbb{Z}$};
  \node[font=\scriptsize, black!60, anchor=west] at (4.95,-0.42) {$\inf P = 0$};
  % multiples of h = 3 - 2 sqrt 2 landing in (x - eps, x + eps)
  \begin{scope}[yshift=-0.95cm]
    \def\h{0.171573}
    \draw[black!40, thin] (-0.4,0) -- (4.85,0);
    \fill[figR!15] (2.69,-0.14) rectangle (3.07,0.14);
    \draw[figR, thick] (2.69,-0.14) -- (2.69,0.14) (3.07,-0.14) -- (3.07,0.14);
    \foreach \q in {0,...,27} \draw[black!60, thin] ({\q*\h},-0.07) -- ({\q*\h},0.07);
    \fill[figR] ({16*\h},0) circle (1.6pt);
    \node[font=\scriptsize, figR, anchor=north] at (2.88,-0.16) {$(x-\varepsilon,\, x+\varepsilon)$};
    \node[font=\scriptsize, black!60, anchor=west] at (4.95,0) {multiples $qh$, \ $h = 3 - 2\sqrt2$};
  \end{scope}
\end{tikzpicture}

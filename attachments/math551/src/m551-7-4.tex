% Figure from MATH 551 notes, section 7. Compile with the notes preamble.
\begin{tikzpicture}[x=5cm, y=0.5cm]
  % ---- dyadic: lines nest ----
  \node[font=\scriptsize, figB, anchor=west] at (-0.1,1.1) {dyadic: every line of one scale reappears at the next};
  \foreach \k/\n/\l in {0/1/1, 1/2/\tfrac12, 2/4/\tfrac14, 3/8/\tfrac18} {
    \pgfmathsetmacro{\y}{-\k}
    \draw[black!60, thin] (0,\y) -- (1,\y);
    \foreach \i in {0,...,\n} \draw[figB, thick] ({\i/\n},\y-0.22) -- ({\i/\n},\y+0.22);
    \node[font=\scriptsize, anchor=east] at (-0.03,\y) {$\l$};
  }
  \foreach \t in {0,0.5,1} \draw[figB!50, thin, densely dotted] (\t,0) -- (\t,-3);
  \draw[figB!50, thin, densely dotted] (0.25,-2) -- (0.25,-3) (0.75,-2) -- (0.75,-3);
  % ---- halves vs thirds ----
  \begin{scope}[yshift=-3.3cm]
    \node[font=\scriptsize, figR, anchor=west] at (-0.1,1.7) {lengths $\tfrac12$ and $\tfrac13$: the lines do not align};
    \draw[black!60, thin] (0,0) -- (1,0);
    \foreach \t in {0,0.5,1} \draw[black!70, thick] (\t,-0.22) -- (\t,0.22);
    \node[font=\scriptsize, anchor=east] at (-0.03,0) {$\tfrac12$};
    \draw[black!60, thin] (0,-1) -- (1,-1);
    \foreach \t in {0,0.3333,0.6667,1} \draw[black!70, thick] (\t,-1.22) -- (\t,-0.78);
    \node[font=\scriptsize, anchor=east] at (-0.03,-1) {$\tfrac13$};
    \foreach \t/\l in {0/0, 0.3333/\tfrac13, 0.5/\tfrac12, 0.6667/\tfrac23, 1/1} \node[font=\scriptsize, below] at (\t,-1.3) {$\l$};
    % overlapping but not nested
    \draw[figB, line width=2.2pt] (0,0) -- (0.5,0);
    \draw[figR, line width=2.2pt] (0.3333,-1) -- (0.6667,-1);
    \fill[figR!12] (0.3333,-1) rectangle (0.5,0);
    \node[font=\scriptsize, figB, anchor=south] at (0.25,0.2) {$(0,\tfrac12]$};
    \node[font=\scriptsize, figR, anchor=west] at (0.69,-0.55) {$(\tfrac13,\tfrac23]$};
  \end{scope}
\end{tikzpicture}

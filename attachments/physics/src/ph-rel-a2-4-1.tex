% Figure for Relativity §A2.4 (spacetime diagram in the frame S of two observers: a source moving at beta = 0.6 emits a flash every unit of its proper time; A, behind it, receives them every 2 units, B, ahead of it, every 0.5 units).
\begin{tikzpicture}[>={Stealth[length=2.2mm]}, font=\small, scale=0.95]
  \draw[->, thin, black!55] (-0.4,0) -- (5.3,0) node[right] {$x$};
  \draw[->, thin, black!55] (0,-0.2) -- (0,7.0) node[above] {$ct$};
  % observers' worldlines
  \draw[black, thick] (0,0) -- (0,6.8);
  \draw[black, thick] (4.5,0) -- (4.5,6.8);
  \node[font=\scriptsize, below] at (0,-0.2) {$A$};
  \node[font=\scriptsize, below] at (4.5,-0.2) {$B$};
  % source worldline, beta = 0.6
  \draw[figO, very thick] (0,0) -- (3.6,6);
  \node[figO, font=\scriptsize, anchor=east, align=right] at (2.95,5.3) {source\\ $v = 0.6c$};
  % flashes at proper times 0..4: (0.75n, 1.25n)
  \foreach \n in {0,1,2,3} {
    \draw[figR, dashed] ({0.75*\n},{1.25*\n}) -- (0,{2*\n});
    \fill[figR] (0,{2*\n}) circle (1.6pt);
  }
  \foreach \n in {0,1,2,3,4} {
    \draw[figB, dashed] ({0.75*\n},{1.25*\n}) -- (4.5,{4.5+0.5*\n});
    \fill[figB] (4.5,{4.5+0.5*\n}) circle (1.6pt);
    \fill[figO] ({0.75*\n},{1.25*\n}) circle (1.8pt);
  }
  \node[figO, font=\scriptsize, anchor=east] at (0.62,1.12) {$\tau = 1$};
  \node[figO, font=\scriptsize, anchor=east] at (1.37,2.37) {$\tau = 2$};
  \node[figR, font=\scriptsize, anchor=east, align=right] at (-0.15,3.0) {received every\\ $\Delta t = 2$};
  \node[figB, font=\scriptsize, anchor=west, align=left] at (4.65,5.5) {received every\\ $\Delta t = 0.5$};
\end{tikzpicture}

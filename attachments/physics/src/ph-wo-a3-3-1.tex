% Figure for Oscillations, Waves and Optics §A3.3 (displacement amplitude of the lowest standing-wave modes in a pipe open at both ends and in a pipe closed at one end).
\begin{tikzpicture}[font=\small]
  \def\L{4.2}\def\H{0.42}
  % left column: open at both ends, s ~ cos(n pi x / L)
  \node[font=\small] at (\L/2,0.95) {open at both ends};
  \foreach \n/\row in {1/0,2/1,3/2} {
    \begin{scope}[yshift={-\row*1.45cm}]
      \draw[black!55, thin] (0,\H+0.08) -- (\L,\H+0.08);
      \draw[black!55, thin] (0,-\H-0.08) -- (\L,-\H-0.08);
      \draw[figB, thick, domain=0:\L, samples=90] plot (\x, {\H*cos(deg(\n*pi*\x/\L))});
      \draw[figB, thick, domain=0:\L, samples=90] plot (\x, {-\H*cos(deg(\n*pi*\x/\L))});
      \foreach \m in {1,...,\n} { \fill[figR] ({(\m-0.5)*\L/\n},0) circle (1.6pt); }
      \node[anchor=west, font=\scriptsize] at (\L+0.15,0.12) {$n=\n$};
      \node[anchor=west, font=\scriptsize] at (\L+0.15,-0.22) {$f_\n = \n\,\tfrac{v}{2L}$};
    \end{scope}
  }
  % right column: closed at x = 0, open at x = L, s ~ sin(n pi x / 2L), n odd
  \begin{scope}[xshift=6.6cm]
    \node[font=\small] at (\L/2,0.95) {closed at left end};
    \foreach \n/\row in {1/0,3/1,5/2} {
      \begin{scope}[yshift={-\row*1.45cm}]
        \draw[black!55, thin] (0,\H+0.08) -- (\L,\H+0.08);
        \draw[black!55, thin] (0,-\H-0.08) -- (\L,-\H-0.08);
        \draw[black, line width=2pt] (0,-\H-0.08) -- (0,\H+0.08);
        \draw[figB, thick, domain=0:\L, samples=90] plot (\x, {\H*sin(deg(\n*pi*\x/(2*\L)))});
        \draw[figB, thick, domain=0:\L, samples=90] plot (\x, {-\H*sin(deg(\n*pi*\x/(2*\L)))});
        \pgfmathtruncatemacro\nn{(\n+1)/2}
        \foreach \m in {1,...,\nn} { \fill[figR] ({2*(\m-1)*\L/\n},0) circle (1.6pt); }
        \node[anchor=west, font=\scriptsize] at (\L+0.15,0.12) {$n=\n$};
        \node[anchor=west, font=\scriptsize] at (\L+0.15,-0.22) {$f_\n = \n\,\tfrac{v}{4L}$};
      \end{scope}
    }
  \end{scope}
  \draw[black!55, |<->|] (0,-3.55) -- node[below, font=\scriptsize] {$L$} (\L,-3.55);
  \draw[black!55, |<->|] (6.6,-3.55) -- node[below, font=\scriptsize] {$L$} (6.6+\L,-3.55);
\end{tikzpicture}

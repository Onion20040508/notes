\begin{tikzpicture}[>={Stealth[length=2.2mm]}, font=\small]
  \def\D{2.0}
  % left panel: clock at rest in S'
  \draw[very thick, black!70] (-0.6,0) -- (0.6,0);
  \draw[very thick, black!70] (-0.6,\D) -- (0.6,\D);
  \draw[->, thick, figB] (-0.12,0.05) -- (-0.12,{\D-0.05});

  \node[font=\scriptsize, align=center] at (0,-1.2) {in $S'$ (clock at rest):\\ $\Delta\tau = 2D/c$};
  % right panel: clock moving in S (beta = 0.6: L = 1.5, s = 2.5)
  \begin{scope}[xshift=3.2cm]
    \def\L{1.5}

    \draw[<->, thin, black!55] (0,-0.25) -- (\L,-0.25) node[midway, below, font=\scriptsize] {$v\,\Delta t/2$};

  \end{scope}

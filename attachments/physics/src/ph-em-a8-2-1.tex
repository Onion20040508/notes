% Figure for Electromagnetism §A8.2: left, field lines of a long straight wire (current out of the page), circles whose spacing grows with r since B is proportional to 1/r; right, two parallel currents attract.
\begin{tikzpicture}[>={Stealth[length=2.0mm]}, font=\small]
  % left: field of a wire, radii in geometric progression so that line density ~ 1/r
  \foreach \k in {0,...,4} {
    \pgfmathsetmacro{\r}{0.32*1.5^\k}
    \draw[figB, thick, postaction={decorate}, decoration={markings, mark=at position 0.125 with {\arrow{Stealth[length=1.8mm]}}, mark=at position 0.625 with {\arrow{Stealth[length=1.8mm]}}}] (\r,0) arc (0:360:\r);
  }
  \filldraw[fill=white, draw=black, thick] (0,0) circle (0.13); \fill (0,0) circle (0.04);
  \node[figB] at (1.9,1.55) {$\vec B$};
  % right: two parallel wires
  \begin{scope}[xshift=5.4cm]
    \draw[figB!50, thin] (0,0) circle (1.6);
    \filldraw[fill=white, draw=black, thick] (0,0) circle (0.13); \fill (0,0) circle (0.04);
    \filldraw[fill=white, draw=black, thick] (1.6,0) circle (0.13); \fill (1.6,0) circle (0.04);
    \node[font=\scriptsize] at (0,-0.38) {$I_1$};
    \node[font=\scriptsize] at (1.6,-0.38) {$I_2$};
    \draw[->, very thick, figB] (1.6,0.16) -- (1.6,1.0) node[right] {$\vec B_1$};
    \draw[->, very thick, figR] (1.45,0) -- (0.85,0);
    \node[figR, font=\scriptsize] at (1.1,0.22) {$\vec F_2$};
    \draw[->, very thick, figR] (0.15,0) -- (0.75,0);
    \node[figR, font=\scriptsize] at (0.45,-0.24) {$\vec F_1$};
    \draw[|<->|, black!55, thin] (0,-0.75) -- (1.6,-0.75) node[midway, below, font=\scriptsize] {$r$};
  \end{scope}
\end{tikzpicture}

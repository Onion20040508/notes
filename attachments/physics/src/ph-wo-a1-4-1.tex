% Figure for Oscillations, Waves and Optics §A1.4: (a) forces on a simple pendulum, (b) a physical pendulum with the weight at the centre of mass and its lever arm about the pivot.
\begin{tikzpicture}[>={Stealth[length=2.2mm]}, font=\small]
  % (a) simple pendulum
  \def\th{30}\def\L{3.4}
  \fill[pattern=north east lines, pattern color=black!45] (-0.6,0) rectangle (0.6,0.18);
  \draw[black!60, thin] (-0.6,0) -- (0.6,0);
  \draw[black!45, thin, dashed] (0,0) -- (0,-\L-0.3);
  \draw[black!70] (0,0) -- ({\L*sin(\th)},{-\L*cos(\th)});
  \node[font=\scriptsize, black!75, anchor=east] at ({0.5*\L*sin(\th)-0.05},{-0.5*\L*cos(\th)}) {$L$};
  \draw[black!55, thin] (0,-1.0) arc (-90:{-90+\th}:1.0);
  \node[font=\scriptsize, black!75] at ({1.25*sin(\th/2)},{-1.25*cos(\th/2)}) {$\theta$};
  \draw[black!45, densely dotted] (0,-\L) arc (-90:{-90+\th}:\L);
  \node[font=\scriptsize, black!60, anchor=north west] at (0.05,{-\L-0.05}) {$s = L\theta$};
  \coordinate (B) at ({\L*sin(\th)},{-\L*cos(\th)});
  \draw[->, thick, black!55] (B) -- ++({-1.0*sin(\th)},{1.0*cos(\th)}) node[right, font=\scriptsize, xshift=1pt] {$\vec T$};
  \draw[->, thick, figB] (B) -- ++(0,-1.7) node[below, font=\scriptsize] {$m\vec g$};
  \draw[->, thick, figR] (B) -- ++({-1.7*sin(\th)*cos(\th)},{-1.7*sin(\th)*sin(\th)}) node[above left, font=\scriptsize, xshift=2pt] {$mg\sin\theta$};
  \draw[->, black!55, dashed] (B) -- ++({1.7*cos(\th)*sin(\th)},{-1.7*cos(\th)*cos(\th)}) node[right, font=\scriptsize] {$mg\cos\theta$};
  \fill[figB] (B) circle (3pt);
  \node[font=\small] at (0,-5.2) {(a)};
  % (b) physical pendulum
  \begin{scope}[xshift=6.4cm]
    \def\ph{25}\def\d{2.0}
    \begin{scope}[rotate=\ph]
      \draw[thick, figB, fill=figB!8] (0,0.35) .. controls (0.9,0.4) and (0.8,-1.6) .. (0.5,-2.6) .. controls (0.2,-3.3) and (-0.7,-3.1) .. (-0.7,-2.2) .. controls (-0.75,-1.3) and (-0.8,0.3) .. (0,0.35) -- cycle;
    \end{scope}
    \draw[black, fill=white] (0,0) circle (2.2pt);
    \node[font=\scriptsize, anchor=east] at (-0.1,-0.05) {$O$};
    \draw[black!45, thin, dashed] (0,0) -- (0,-3.3);
    \coordinate (C) at ({\d*sin(\ph)},{-\d*cos(\ph)});
    \draw[black!70] (0,0) -- (C);
    \node[font=\scriptsize, black!75, anchor=south west] at ({0.55*\d*sin(\ph)+0.02},{-0.55*\d*cos(\ph)}) {$L$};
    \draw[black!55, thin] (0,-0.8) arc (-90:{-90+\ph}:0.8);
    \node[font=\scriptsize, black!75] at ({1.05*sin(\ph/2)},{-1.05*cos(\ph/2)}) {$\theta$};
    \fill[black] (C) circle (1.8pt);
    \node[font=\scriptsize, anchor=west] at ($(C)+(0.08,0.05)$) {CM};
    \draw[->, thick, figB] (C) -- ++(0,-1.4) node[below, font=\scriptsize] {$m\vec g$};
    \draw[black!45, thin, densely dotted] (C) -- ({\d*sin(\ph)},0);
    \draw[thick, figR] (0,0) -- ({\d*sin(\ph)},0);
    \node[figR, font=\scriptsize, anchor=south] at ({0.5*\d*sin(\ph)+0.25},0.2) {$L\sin\theta$};
    \node[font=\small] at (0,-5.2) {(b)};
  \end{scope}
\end{tikzpicture}

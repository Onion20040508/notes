% Figure for Electromagnetism §A11.1: voltage and current versus time, and the phasor diagrams, for a resistor, a capacitor and an inductor across an ac source.
\begin{tikzpicture}[>={Stealth[length=2.2mm]}, font=\small]
  \foreach \k/\ph/\name/\txt in {0/0/resistor/{in phase}, 1/90/capacitor/{$i$ leads $v$ by $\pi/2$}, 2/-90/inductor/{$i$ lags $v$ by $\pi/2$}} {
    \begin{axis}[nfig, at={(\k*4.2cm,0)}, width=4.4cm, height=3.4cm,
        xmin=0, xmax=12.9, ymin=-1.3, ymax=1.3,
        xtick={0,3.1416,6.2832,9.4248,12.566}, xticklabels={$0$,$\pi$,$2\pi$,$3\pi$,$4\pi$},
        ytick=\empty, xlabel={$\omega t$}, xlabel style={font=\scriptsize, yshift=4pt},
        title={\name: \txt}, title style={font=\scriptsize, yshift=-2pt}]
      \addplot[black!30, thin, domain=0:12.9, samples=2] {0};
      \addplot[figB, domain=0:12.6, samples=120] {sin(deg(x))};
      \addplot[figR, domain=0:12.6, samples=120, dashed] {0.65*sin(deg(x)+\ph)};
    \end{axis}
    % phasor diagram below, drawn at the instant omega t = 30 deg
    \begin{scope}[shift={(\k*4.2cm+1.6cm,-2.4cm)}]
      \draw[thin, black!55] (-1.1,0) -- (1.2,0);
      \draw[->, thin, black!55] (0,-1.1) -- (0,1.2);
      \draw[->, very thick, figB] (0,0) -- (30:1.0) node[anchor=south west, font=\scriptsize, inner sep=1pt] {$V_0$};
      \ifnum\k=0
        \draw[->, very thick, figR, shorten >=0pt] ([shift={(0,-0.07)}]0,0) -- ([shift={(0,-0.07)}]30:0.65) node[anchor=north west, font=\scriptsize, inner sep=1pt] {$I_0$};
      \else
        \draw[->, very thick, figR] (0,0) -- ({30+\ph}:0.65) node[font=\scriptsize, anchor={180+30+\ph}, inner sep=1pt] {$I_0$};
      \fi
      \draw[black!55, thin] (0.4,0) arc (0:30:0.4);
      \node[font=\scriptsize, black!70] at (0.75,-0.17) {$\omega t$};
    \end{scope}
  }
\end{tikzpicture}

  \newcommand{\rwall}[2]{\fill[pattern=north east lines, pattern color=black!55] (#1,#2-0.25) rectangle ++(0.25,\h+0.5); \draw[black!70, thick] (#1,#2-0.25) -- ++(0,\h+0.5);}

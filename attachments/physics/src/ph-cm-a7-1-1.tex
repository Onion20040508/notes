% Figure for Classical Mechanics §A7.1 (free-body diagram of a uniform ladder against a frictionless wall; pivot at the foot).
\begin{tikzpicture}[>={Stealth[length=2.4mm]}, font=\small]
  \def\b{53}
  \def\L{4.2}
  \coordinate (A) at (0,0);
  \coordinate (B) at ({\L*cos(\b)},{\L*sin(\b)});
  \coordinate (M) at ({0.5*\L*cos(\b)},{0.5*\L*sin(\b)});
  % floor and wall
  \fill[pattern=north east lines, pattern color=black!35] (-1.2,-0.18) rectangle ({\L*cos(\b)+0.25},0);
  \draw[thick, black!70] (-1.2,0) -- ({\L*cos(\b)},0);
  \fill[pattern=north east lines, pattern color=black!35] ({\L*cos(\b)},0) rectangle ({\L*cos(\b)+0.25},4.0);
  \draw[thick, black!70] ({\L*cos(\b)},0) -- ({\L*cos(\b)},4.0);
  % ladder
  \draw[line width=3pt, black!45] (A) -- (B);
  \draw[black!55, thin] (0.5,0) arc (0:\b:0.5); \node[font=\scriptsize, black!70] at (0.63,0.27) {$\beta$};
  % lever arms
  \draw[black!55, dashed, thin] (M) -- ({0.5*\L*cos(\b)},0);
  \draw[black!55, |<->|, thin] (0,-1.05) -- ({0.5*\L*cos(\b)},-1.05) node[midway, below, font=\scriptsize] {$\tfrac L2\cos\beta$};
  \draw[black!55, dashed, thin] (B) -- (-0.6,{\L*sin(\b)});
  \draw[black!55, |<->|, thin] (-0.6,0) -- (-0.6,{\L*sin(\b)}) node[midway, left, font=\scriptsize] {$L\sin\beta$};
  % forces
  \draw[->, very thick, figB] (A) -- (0,1.7) node[above left, xshift=2pt] {$\vec N$};
  \draw[->, very thick, figB] (A) -- (0.85,0) node[below, yshift=-4pt] {$\vec f$};
  \draw[->, very thick, figB] (M) -- ++(0,-1.3) node[right] {$\vec w$};
  \draw[->, very thick, figR] (B) -- ++(-1.1,0) node[above, xshift=2pt] {$\vec F$};
  \fill (A) circle (1.8pt) node[below left, font=\scriptsize] {pivot};
\end{tikzpicture}

% Figure for Classical Mechanics §A3.9 (Theorem: ideal banking angle; forces on a car on a frictionless banked curve, seen from behind).
\begin{tikzpicture}[>={Stealth[length=2.4mm]}, font=\small]
  \def\th{25}
  % banked road (rises to the right; centre of the curve to the left)
  \fill[black!8] (-3.2,{-3.2*tan(\th)}) -- (3.2,{3.2*tan(\th)}) -- (3.2,-2.5) -- (-3.2,-2.5) -- cycle;
  \draw[black!55, thin] (-3.2,{-3.2*tan(\th)}) -- (3.2,{3.2*tan(\th)});
  \draw[black!40, thin, dashed] (-3.2,{-3.2*tan(\th)}) -- (-1.4,{-3.2*tan(\th)});
  \draw[black!55, thin] (-2.1,{-3.2*tan(\th)}) arc (0:\th:1.1);
  \node[font=\scriptsize, black!70] at (-1.75,{-3.2*tan(\th)+0.2}) {$\theta$};
  % car body
  \begin{scope}[rotate=\th]
    \draw[thick, black!70, fill=white] (-0.8,0) rectangle (0.8,0.7);
    \coordinate (C) at (0,0.35);
  \end{scope}
  % normal force, perpendicular to road
  \draw[->, very thick, figB] (C) -- ++({90+\th}:2.6) coordinate (Ntip) node[above left] {$\vec N$};
  % components of N
  \draw[->, thick, figB, dashed] (C) -- ++(0,{2.6*cos(\th)}) node[right, font=\scriptsize] {$N\cos\theta$};
  \draw[->, very thick, figR] (C) -- ++({-2.6*sin(\th)},0) node[left, font=\scriptsize, align=right] {$N\sin\theta = mv^2/r$};
  % weight
  \draw[->, very thick, figB] (C) -- ++(0,{-2.6*cos(\th)}) node[below right, xshift=-2pt] {$m\vec g$};
  % angle between N and vertical
  \draw[black!55, thin] ($(C)+(0,1.1)$) arc (90:90+\th:1.1);
  \node[font=\scriptsize, black!70] at ($(C)+(-0.25,1.4)$) {$\theta$};
  \fill (C) circle (1.8pt);
  % toward the centre
  \draw[->, thin, black!55] (-1.6,3.5) -- (-3.0,3.5) node[left, font=\scriptsize] {to centre};
\end{tikzpicture}

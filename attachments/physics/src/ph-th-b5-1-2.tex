% Figure for Thermal Physics §B5.1 (nitrogen at 300 K: speed distribution of the molecules in the gas, proportional to v^2 exp(-v^2/v_p^2), against that of the molecules effusing through a small hole, proportional to v^3 exp(-v^2/v_p^2), both normalized; means 476 m/s and 561 m/s marked; computed).
\begin{tikzpicture}[font=\small,
  declare function={vp=422.0; fg(\v)=1000*4*\v^2/(sqrt(pi)*vp^3)*exp(-(\v/vp)^2); fe(\v)=1000*2*\v^3/(vp^4)*exp(-(\v/vp)^2);}]
  \begin{axis}[nfig, width=10cm, height=6cm, xmin=0, xmax=1500, ymin=0, ymax=2.3,
      xlabel={$v$ (m/s)}, ylabel={distribution ($10^{-3}$ s/m)},
      xtick={0,500,1000,1500}, ytick={0.5,1,1.5,2},
      scaled x ticks=false, x tick label style={/pgf/number format/1000 sep={}},
      xlabel style={at={(axis description cs:1.02,0)}, anchor=west},
      ylabel style={rotate=-90, at={(axis description cs:0,1)}, anchor=south west, xshift=-12pt}]
    \addplot[figB, domain=0:1500, samples=200] {fg(x)};
    \addplot[figR, domain=0:1500, samples=200] {fe(x)};
    \draw[figB, dashed] (axis cs:476,0) -- (axis cs:476,{fg(476)});
    \draw[figR, dashed] (axis cs:561,0) -- (axis cs:561,{fe(561)});
    \node[figB, font=\scriptsize, anchor=west] at (axis cs:820,1.72) {in the gas, $\propto v^2e^{-mv^2/2k_BT}$};

  \end{axis}
\end{tikzpicture}

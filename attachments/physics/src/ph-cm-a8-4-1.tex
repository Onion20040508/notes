% Figure for Classical Mechanics §A8.4: tidal field of the Moon at Earth's surface (gravitational field minus its value at Earth's centre), proportional to (2x, -y) with the Moon on the +x axis; the exaggerated tidal bulge is dashed.
\begin{tikzpicture}[font=\small, >={Stealth[length=1.9mm]}]
  \def\R{1.3}
  % exaggerated bulge
  \draw[figB!60, dashed, thick] (0,0) ellipse (1.72 and 1.12);
  % Earth
  \fill[figB!12] (0,0) circle (\R);
  \draw[figB, thick] (0,0) circle (\R);
  \fill[figB] (0,0) circle (1.3pt);
  \node[figB, font=\scriptsize, anchor=north] at (0,-0.05) {Earth};
  % tidal arrows
  \foreach \t in {0,30,...,330} {
    \draw[->, figR, thick] ({\R*cos(\t)},{\R*sin(\t)}) -- ++({0.42*2*cos(\t)},{-0.42*sin(\t)});
  }
  % Moon direction
  \draw[->, black!55, thin] (3.0,0) -- (4.3,0) node[right, font=\scriptsize, black!70] {to the Moon};
  \node[font=\scriptsize, figR, anchor=south] at (2.3,0.12) {near side};
  \node[font=\scriptsize, figR, anchor=south] at (-2.3,0.12) {far side};
\end{tikzpicture}

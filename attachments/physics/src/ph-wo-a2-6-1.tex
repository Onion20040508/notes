% Figure for Oscillations, Waves and Optics §A2.6 (top: two identical sinusoids travelling in opposite directions and their sum at one instant; bottom: the sum 2A sin(kx) cos(wt) at several instants, with fixed nodes and antinodes).
\begin{tikzpicture}[font=\small, >={Stealth[length=2.2mm]}]
  \begin{axis}[nfig, name=top, width=10.5cm, height=3.8cm, xmin=0, xmax=2.08, ymin=-2.2, ymax=2.2,
      axis lines=middle, xtick=\empty, ytick={-2,-1,1,2}, yticklabels={$-2A$,$-A$,$A$,$2A$}, clip=false]
    \addplot[figB, thick, domain=0:2, samples=120] {sin(deg(2*pi*x) - 60)};
    \addplot[figG, thick, dashed, domain=0:2, samples=120] {sin(deg(2*pi*x) + 60)};
    \addplot[figR, very thick, domain=0:2, samples=120] {2*sin(deg(2*pi*x))*cos(60)};
    \node[font=\scriptsize, anchor=south] at (axis cs:1.0,2.25) {\textcolor{figB}{$A\sin(kx - \omega t)$ (to the right)}, \quad \textcolor{figG}{$A\sin(kx + \omega t)$ (to the left, dashed)}, \quad \textcolor{figR}{sum at $\omega t = \pi/3$}};
  \end{axis}
  \begin{axis}[nfig, at={(top.south)}, anchor=north, yshift=-0.2cm, width=10.5cm, height=3.8cm, xmin=0, xmax=2.08, ymin=-2.2, ymax=2.2,

    }
    \addplot[only marks, mark=*, mark size=1.8pt, black] coordinates {(0,0) (0.5,0) (1,0) (1.5,0) (2,0)};

  \end{axis}

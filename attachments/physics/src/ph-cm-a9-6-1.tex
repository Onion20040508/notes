  \foreach \xs/\lab/\vis in {0/{ideal fluid ($\eta = 0$)}/0, 5.2/{viscous fluid, laminar}/1} {

      \fill[black!25] (-0.3,1.0) rectangle (3.7,1.12);
      \fill[black!25] (-0.3,-1.0) rectangle (3.7,-1.12);
      \draw[black!70] (-0.3,1.0) -- (3.7,1.0);

  % ideal: uniform arrows
  \foreach \y in {-0.84,-0.6,...,0.85} \draw[->, figB, thick] (0.4,\y) -- (2.4,\y);

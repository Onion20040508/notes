% Figure for Oscillations, Waves and Optics §A2.5 (snapshots of two sinusoidal waves of equal amplitude, wavelength and frequency that differ by a phase shift phi, and their sum 2A cos(phi/2) sin(kx - wt + phi/2), for phi = 0, 2pi/3 and pi).
\begin{tikzpicture}[font=\small]
  \foreach \ph/\lab/\j in {0/{\varphi = 0}/0, 120/{\varphi = 2\pi/3}/1, 180/{\varphi = \pi}/2} {
    \begin{axis}[nfig, at={(0,-\j*2.55cm)}, width=10cm, height=3.4cm, xmin=0, xmax=2.05, ymin=-2.25, ymax=2.25,
        axis lines=middle, xtick=\empty, ytick={-2,-1,1,2}, yticklabels={$-2A$,$-A$,$A$,$2A$}, clip=false]
      \addplot[figB, thick, domain=0:2, samples=120] {sin(deg(2*pi*x))};
      \addplot[figG, thick, dashed, domain=0:2, samples=120] {sin(deg(2*pi*x) + \ph)};
      \addplot[figR, very thick, domain=0:2, samples=120] {sin(deg(2*pi*x)) + sin(deg(2*pi*x) + \ph)};
      \node[font=\scriptsize, anchor=west] at (axis cs:2.1,0) {$\lab$};
    \end{axis}
  }
  \node[font=\scriptsize, anchor=south] at (4.3,2.35) {\textcolor{figB}{$y_1 = A\sin(kx - \omega t)$}, \quad \textcolor{figG}{$y_2 = A\sin(kx - \omega t + \varphi)$ (dashed)}, \quad \textcolor{figR}{$y_1 + y_2$}};
\end{tikzpicture}

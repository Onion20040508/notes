% Figure for Fourier Series and PDEs §54.
\begin{tikzpicture}[declare function={kap=sqrt(pi); al=0.5;
    up(\x,\p) = 1 + al*exp(-kap*\x)*sin(deg(\p - kap*\x));}]
  \begin{axis}[nfig, width=11cm, height=6cm, xmin=0, xmax=4, ymin=0.4, ymax=1.6, clip=true,
      xlabel={$x$}, ylabel={$u$}, xtick={0,1,2,3,4}, ytick={0.5,1,1.5},
      xlabel style={anchor=north}, ylabel style={anchor=south}]
    \addplot[black!45, dashed, thin, samples=100, domain=0:4] {1 + al*exp(-kap*x)};
    \addplot[black!45, dashed, thin, samples=100, domain=0:4] {1 - al*exp(-kap*x)};
    \addplot[figB, samples=150, domain=0:4] {up(x,0)};
    \addplot[figR, samples=150, domain=0:4] {up(x,pi/2)};
    \addplot[figG, samples=150, domain=0:4] {up(x,pi)};
    \addplot[figO, samples=150, domain=0:4] {up(x,3*pi/2)};
  \end{axis}
  \begin{scope}[font=\scriptsize, shift={(10.0cm,3.9cm)}]
    \draw[figB, line width=0.9pt] (0,0) -- (0.5,0) node[right, black] {$\omega t = 0$};
    \draw[figR, line width=0.9pt] (0,-0.4) -- (0.5,-0.4) node[right, black] {$\omega t = \pi/2$};
    \draw[figG, line width=0.9pt] (0,-0.8) -- (0.5,-0.8) node[right, black] {$\omega t = \pi$};
    \draw[figO, line width=0.9pt] (0,-1.2) -- (0.5,-1.2) node[right, black] {$\omega t = 3\pi/2$};
    \draw[black!45, dashed] (0,-1.6) -- (0.5,-1.6) node[right, black] {$1 \pm \alpha e^{-\sqrt{\omega/2}\,x}$};
  \end{scope}
\end{tikzpicture}

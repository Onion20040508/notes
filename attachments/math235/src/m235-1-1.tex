% Figure for Applied Linear Algebra §1.
\begin{tikzpicture}[x=0.75cm, y=0.75cm, >=Stealth, font=\scriptsize]
  % panel (a): one solution  x+y=2, x-y=1
  \begin{scope}
    \draw[->, black!60] (-0.6,0) -- (3.4,0) node[right] {$x$};
    \draw[->, black!60] (0,-1.4) -- (0,3.2) node[above] {$y$};
    \draw[figB, thick] (-0.6,2.6) -- (3.0,-1.0) node[right] {$x+y=2$};
    \draw[figR, thick] (-0.3,-1.3) -- (3.2,2.2) node[above] {$x-y=1$};
    \fill (1.5,0.5) circle (2pt);
    \node[above left=1pt] at (1.5,0.5) {$(\tfrac32,\tfrac12)$};
    \node[font=\small] at (1.4,-2.1) {(a) exactly one solution};
  \end{scope}
  % panel (b): no solution  x+y=2, x+y=3
  \begin{scope}[xshift=5.4cm]
    \draw[->, black!60] (-0.6,0) -- (3.4,0) node[right] {$x$};
    \draw[->, black!60] (0,-1.4) -- (0,3.2) node[above] {$y$};
    \draw[figB, thick] (-0.6,2.6) -- (3.0,-1.0) node[below] {$x+y=2$};
    \draw[figR, thick] (0.0,3.0) -- (3.4,-0.4) node[right] {$x+y=3$};
    \node[font=\small] at (1.4,-2.1) {(b) no solution};
  \end{scope}
  % panel (c): infinitely many  x+y=2, 2x+2y=4
  \begin{scope}[xshift=10.8cm]
    \draw[->, black!60] (-0.6,0) -- (3.4,0) node[right] {$x$};
    \draw[->, black!60] (0,-1.4) -- (0,3.2) node[above] {$y$};
    \draw[figR, line width=3pt, opacity=0.35] (-0.6,2.6) -- (3.0,-1.0);
    \draw[figB, thick] (-0.6,2.6) -- (3.0,-1.0);
    \node[figB, right, align=left] at (1.3,1.3) {$x+y=2$\\ \textcolor{figR}{$2x+2y=4$}};
    \node[font=\small] at (1.4,-2.1) {(c) infinitely many};
  \end{scope}
\end{tikzpicture}

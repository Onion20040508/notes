% Figure from MATH 452 notes, section 15. Compile with the notes preamble.
% D = ellipse centered (2,1.5), semi-axes 1.75 and 1.3, on the grid [0,4]x[0,3].
% A square is inner if all four corners lie in D (D convex); it meets D iff the point of the
% square nearest the center (clamped coordinates) lies in D.
\begin{tikzpicture}[scale=0.88]
\foreach \s/\n/\m/\sh in {1/3/2/0, 0.5/7/5/6.3}{
\begin{scope}[shift={(\sh,0)}]
\foreach \i in {0,...,\n}{ \foreach \j in {0,...,\m}{
  \pgfmathsetmacro{\xa}{\i*\s} \pgfmathsetmacro{\xb}{(\i+1)*\s}
  \pgfmathsetmacro{\ya}{\j*\s} \pgfmathsetmacro{\yb}{(\j+1)*\s}
  \pgfmathsetmacro{\caa}{((\xa-2)/1.75)^2+((\ya-1.5)/1.3)^2}
  \pgfmathsetmacro{\cab}{((\xa-2)/1.75)^2+((\yb-1.5)/1.3)^2}
  \pgfmathsetmacro{\cba}{((\xb-2)/1.75)^2+((\ya-1.5)/1.3)^2}
  \pgfmathsetmacro{\cbb}{((\xb-2)/1.75)^2+((\yb-1.5)/1.3)^2}
  \pgfmathtruncatemacro{\allin}{((\caa<=1)&&(\cab<=1)&&(\cba<=1)&&(\cbb<=1)) ? 1 : 0}
  \pgfmathsetmacro{\px}{max(\xa,min(2,\xb))} \pgfmathsetmacro{\py}{max(\ya,min(1.5,\yb))}
  \pgfmathsetmacro{\cc}{((\px-2)/1.75)^2+((\py-1.5)/1.3)^2}
  \pgfmathtruncatemacro{\meets}{(\cc<=1) ? 1 : 0}
  \ifnum\allin>0
    \fill[figB!25] (\xa,\ya) rectangle (\xb,\yb);
  \else
    \ifnum\meets>0
      \fill[figR!18] (\xa,\ya) rectangle (\xb,\yb);
    \fi
  \fi
}}
\draw[black!35, thin] (0,0) grid[step=\s] (4,3);
\draw[black, thick] (2,1.5) ellipse (1.75 and 1.3);
\node[font=\small, anchor=south west, inner sep=1pt] at ({2+1.75*cos(40)},{1.5+1.3*sin(40)}) {$D$};
\end{scope}}
\node[font=\scriptsize, align=center, anchor=north] at (2,-0.12) {$i = 0$ (side $1$)\\ $A_0^-(D) = 2 \cdot 1 = 2,\ \ A_0^+(D) = 12 \cdot 1 = 12$};
\node[font=\scriptsize, align=center, anchor=north] at (8.3,-0.12) {$i = 1$ (side $\tfrac12$)\\ $A_1^-(D) = 20 \cdot \tfrac14 = 5,\ \ A_1^+(D) = 44 \cdot \tfrac14 = 11$};
\end{tikzpicture}

% Figure from MATH 452 notes, section 15. Compile with the notes preamble.
% Type II primitive x = Psi(u,v) = u + 0.3v + 0.15v^2 (v fixed), then Type I primitive y = Phi(v,x) = v + 0.2x^2 (x fixed).
\begin{tikzpicture}[scale=1.0,
  declare function={Ps(\u,\v)=\u+0.3*\v+0.15*\v*\v; Ph(\v,\x)=\v+0.2*\x*\x;}]
% ---------- (u,v) ----------
\begin{scope}
\draw[-stealth, thin] (-0.1,0) -- (1.9,0) node[right, font=\scriptsize] {$u$};
\draw[-stealth, thin] (0,-0.1) -- (0,2.2) node[above, font=\scriptsize] {$v$};
\fill[figB!10] (0.2,0.2) rectangle (1.4,1.4);
\foreach \a in {0.2,0.6,1.0,1.4} \draw[figB!70, thin] (\a,0.2) -- (\a,1.4);
\foreach \b in {0.2,0.6,1.0,1.4} \draw[figG!80!black, thin] (0.2,\b) -- (1.4,\b);
\filldraw[figR] (1.0,1.0) circle (1.2pt);
\node[font=\scriptsize, anchor=north, align=center] at (0.85,-0.1) {grid in $(u,v)$};
\end{scope}
\draw[-stealth, thick] (1.85,1.2) -- (2.55,1.2) node[midway, above, font=\scriptsize] {II};
% ---------- (x,v) ----------
\begin{scope}[shift={(2.8,0)}]
\draw[-stealth, thin] (-0.1,0) -- (2.6,0) node[right, font=\scriptsize] {$x$};
\draw[-stealth, thin] (0,-0.1) -- (0,2.2) node[above, font=\scriptsize] {$v$};
\fill[figB!10] plot[domain=0.2:1.4] ({Ps(\x,0.2)},0.2) -- plot[domain=0.2:1.4, samples=20] ({Ps(1.4,\x)},\x)
  -- plot[domain=1.4:0.2] ({Ps(\x,1.4)},1.4) -- plot[domain=1.4:0.2, samples=20] ({Ps(0.2,\x)},\x) -- cycle;
\foreach \a in {0.2,0.6,1.0,1.4} \draw[figB!70, thin, domain=0.2:1.4, samples=20] plot ({Ps(\a,\x)},\x);
\foreach \b in {0.2,0.6,1.0,1.4} \draw[figG!80!black, thin] ({Ps(0.2,\b)},\b) -- ({Ps(1.4,\b)},\b);
\draw[-stealth, figR, thick] (1.0,1.0) -- ({Ps(1.0,1.0)-0.03},1.0);
\filldraw[fill=white, draw=figR] (1.0,1.0) circle (1.2pt);
\node[font=\scriptsize, anchor=north, align=center] at (1.25,-0.1) {in $(x,v)$: points slide\\ horizontally, $x = \Psi(u,v)$};
\filldraw[figR] ({Ps(1.0,1.0)},1.0) circle (1.2pt);
\end{scope}
\draw[-stealth, thick] (5.5,1.2) -- (6.2,1.2) node[midway, above, font=\scriptsize] {I};
% ---------- (x,y) ----------
\begin{scope}[shift={(6.45,0)}]
\draw[-stealth, thin] (-0.1,0) -- (2.6,0) node[right, font=\scriptsize] {$x$};
\draw[-stealth, thin] (0,-0.1) -- (0,2.2) node[above, font=\scriptsize] {$y$};
\fill[figB!10] plot[domain=0.2:1.4, samples=20] ({Ps(\x,0.2)},{Ph(0.2,Ps(\x,0.2))}) -- plot[domain=0.2:1.4, samples=20] ({Ps(1.4,\x)},{Ph(\x,Ps(1.4,\x))})
  -- plot[domain=1.4:0.2, samples=20] ({Ps(\x,1.4)},{Ph(1.4,Ps(\x,1.4))}) -- plot[domain=1.4:0.2, samples=20] ({Ps(0.2,\x)},{Ph(\x,Ps(0.2,\x))}) -- cycle;
\foreach \a in {0.2,0.6,1.0,1.4} \draw[figB!70, thin, domain=0.2:1.4, samples=20] plot ({Ps(\a,\x)},{Ph(\x,Ps(\a,\x))});
\foreach \b in {0.2,0.6,1.0,1.4} \draw[figG!80!black, thin, domain=0.2:1.4, samples=20] plot ({Ps(\x,\b)},{Ph(\b,Ps(\x,\b))});
\pgfmathsetmacro{\xp}{Ps(1.0,1.0)}
\filldraw[fill=white, draw=figR] (\xp,1.0) circle (1.2pt);
\node[font=\scriptsize, anchor=north, align=center] at (1.25,-0.1) {in $(x,y)$: points slide\\ vertically, $y = \Phi(v,x)$};
\draw[-stealth, figR, thick] (\xp,1.0) -- (\xp,{Ph(1.0,\xp)-0.03});
\filldraw[figR] (\xp,{Ph(1.0,\xp)}) circle (1.2pt);
\end{scope}
\end{tikzpicture}

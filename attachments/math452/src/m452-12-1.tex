\begin{tikzpicture}[scale=1.55]
  \draw[-stealth, thin] (-1.35,0) -- (1.5,0) node[right, font=\scriptsize] {$x$};
  \draw[-stealth, thin] (0,-1.3) -- (0,1.4) node[above, font=\scriptsize] {$y$};
  \draw[figB, thick] (0,0) circle (1);
  \node[font=\small, figB, anchor=north east, inner sep=1pt] at (-0.72,-0.72) {$x^2+y^2=1$};
  % x0 = cos 55, delta = 0.25
  \pgfmathsetmacro{\xo}{cos(55)} \pgfmathsetmacro{\yo}{sin(55)}
  \pgfmathsetmacro{\xa}{\xo-0.25} \pgfmathsetmacro{\xb}{\xo+0.25}
  \draw[black!40, dotted] (\xa,0) -- (\xa,{sqrt(1-\xa*\xa)});
  \draw[black!40, dotted] (\xb,0) -- (\xb,{sqrt(1-\xb*\xb)});
  \draw[black!40, dotted] (\xo,0) -- (\xo,\yo);
  \draw[figR, very thick, domain=\xa:\xb, samples=40] plot (\x, {sqrt(1-\x*\x)});
  % the interval (x0-delta, x0+delta)
  \draw[figR, thick] (\xa,0) -- (\xb,0);
  \filldraw[fill=white, draw=figR] (\xa,0) circle (0.9pt);
  \filldraw[fill=white, draw=figR] (\xb,0) circle (0.9pt);
  \filldraw[black] (\xo,0) circle (0.7pt);
  \node[font=\scriptsize, anchor=north] at (\xo,-0.04) {$x_0$};
  \filldraw[black] (\xo,\yo) circle (0.9pt);
  \node[font=\scriptsize, anchor=south west, inner sep=1.5pt] at ({\xo+0.03},{\yo+0.03}) {$(x_0,y_0)$};
  \node[font=\scriptsize, figR, anchor=west] at ({\xb+0.13},{sqrt(1-\xb*\xb)-0.04}) {$y=f(x)$};
  \draw[decorate, decoration={brace, amplitude=3pt, mirror}, black!55] (\xa,-0.2) -- (\xb,-0.2);
  \node[font=\scriptsize, black!70, anchor=north] at (\xo,-0.27) {$|x-x_0|<\delta$};
\end{tikzpicture}

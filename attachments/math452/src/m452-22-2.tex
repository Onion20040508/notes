% Figure from MATH 452 notes, section 22. Compile with the notes preamble.
\begin{tikzpicture}[scale=1.15]
% vector field (-y,x)/(x^2+y^2): arrows centered at grid points, length 0.3/r (capped), origin skipped
\foreach \x in {-1.8,-1.2,-0.6,0,0.6,1.2,1.8}
  \foreach \y in {-1.8,-1.2,-0.6,0,0.6,1.2,1.8}
    {\pgfmathsetmacro{\rsq}{\x*\x+\y*\y}
     \pgfmathparse{\rsq > 0.1 ? 1 : 0}
     \ifnum\pgfmathresult>0
       \pgfmathsetmacro{\r}{sqrt(\rsq)}
       \pgfmathsetmacro{\len}{min(0.3/\r,0.4)}
       \pgfmathsetmacro{\vx}{-\len*\y/\r}
       \pgfmathsetmacro{\vy}{\len*\x/\r}
       \draw[-stealth, figB!70, thin] ({\x-\vx/2},{\y-\vy/2}) -- ({\x+\vx/2},{\y+\vy/2});
     \fi}
% unit circle gamma with counterclockwise orientation
\draw[thick, figR,
  postaction={decorate},
  decoration={markings,
    mark=at position 0.19 with {\arrow[figR, scale=1.4]{Stealth}},
    mark=at position 0.69 with {\arrow[figR, scale=1.4]{Stealth}}}]
  (0,0) circle (1);
\node[figR, font=\small] at (0.95,0.95) {$\boldsymbol{\gamma}$};
% the removed origin
\filldraw[fill=white, draw=black, thick] (0,0) circle (2pt);
\node[font=\scriptsize, black!70, anchor=south] at (0,0.07) {hole};
\end{tikzpicture}%
